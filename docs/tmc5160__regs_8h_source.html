191/* -----------------------------------------------------------------------
192 * IHOLD_IRUN packing helper
193 * bits [4:0] = IHOLD (standstill current, 0-31)
194 * bits [12:8] = IRUN (run current, 0-31)
195 * bits [19:16] = IHOLDDELAY
196 * --------------------------------------------------------------------- */
197#define TMC_IHOLD_IRUN(ihold, irun, idelay) \
198 (((uint32_t)(ihold) & 0x1Fu) | \
199 (((uint32_t)(irun) & 0x1Fu) << 8) | \
200 (((uint32_t)(idelay)& 0x0Fu) << 16))
